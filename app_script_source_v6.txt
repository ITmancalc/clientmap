function doGet(e){ return route_(e, "GET"); }
function doPost(e){ return route_(e, "POST"); }

function route_(e, method){
  const p = (e && e.parameter) ? e.parameter : {};
  const action = String(p.action || "").toLowerCase() || (method === "POST" ? "save" : "stats");
  const cb = p.callback ? String(p.callback) : "";
  try{
    let data;
    if(action === "ping") data = {ok:true, version:CFG.VERSION, now:new Date().toISOString()};
    else if(action === "search") data = handleSearch_(p);
    else if(action === "get") data = handleGet_(p);
    else if(action === "latest") data = handleLatest_(p);
    else if(action === "history") data = handleHistory_(p);
    else if(action === "save") data = handleSave_(e);
    else if(action === "stats") data = handleStats_(p);
    else data = {ok:false, error:"Unknown action: " + action, version:CFG.VERSION};

    return output_(data, cb);
  }catch(err){
    const out = {ok:false, error:String(err && err.message ? err.message : err), stack:String(err && err.stack ? err.stack : ""), version:CFG.VERSION};
    return output_(out, cb);
  }
}

// Returns the latest row for a company, merged with parsed `payload` JSON (same behavior as handleGet_).
function handleLatest_(p){
  const sh = getDataSheet_();
  const {headers, map} = getHeaderMap_(sh);
  const lastRow = sh.getLastRow();
  if(lastRow < 2) return {ok:false, error:"No data", version:CFG.VERSION};

  const company = String(p.company||"").trim();
  if(!company) return {ok:false, error:"company is required", version:CFG.VERSION};

  const colCompany = map["company"] || map["Компания"] || 1;
  let foundRow = -1;
  if(lastRow >= 2){
    const names = sh.getRange(2,colCompany,lastRow-1,1).getValues();
    for(let i=names.length-1;i>=0;i--){
      if(String(names[i][0]||"").trim() === company){ foundRow = i+2; break; }
    }
  }
  if(foundRow < 0) return {ok:false, error:"Not found", version:CFG.VERSION};

  // Reuse handleGet_ logic by passing row
  return handleGet_({row:foundRow});
}

// Returns all saves for a company as a compact series (for charts).
function handleHistory_(p){
  const company = String(p.company||"").trim();
  if(!company) return {ok:false, error:"company is required", version:CFG.VERSION};
  const limit = Math.max(1, Math.min(1000, Number(p.limit||400)));

  const sh = getDataSheet_();
  const {headers, map} = getHeaderMap_(sh);
  const lastRow = sh.getLastRow();
  if(lastRow < 2) return {ok:true, items:[], company, version:CFG.VERSION};

  const colCompany = map["company"] || map["Компания"] || 1;
  const colTs = findTimestampCol_(headers, map);

  function num_(v){
    const n = Number(v);
    return (isNaN(n) || v === "") ? null : n;
  }

  const idxCols = {
    timestamp: colTs,
    workplaces: map["workplaces"] || map["Workplaces"] || 0,
    servers: map["servers"] || map["Servers"] || 0,
    psi: map["psi2Score"] || map["psi"] || 0,
    tech: map["techIndex"] || 0,
    proc: map["procIndex"] || 0,
    risk: map["riskIndex"] || 0,
    prob: map["probIndex"] || 0,
    ready: map["readyIndex"] || 0,
    coiTotal: map["coiTotal"] || map["coi_total"] || 0,
    recover: map["recoverIndex"] || 0,
    bus: map["busIndex"] || 0,
    obsol: map["obsolIndex"] || 0,
    ai: map["aiIndex"] || 0,
    pain: map["painIndex"] || 0,
    feas: map["feasScore"] || 0,
    risk_grey_pct: map["risk_grey_pct"] || 0,
    risk_lic_pct: map["risk_lic_pct"] || 0,
    risk_ops_pct: map["risk_ops_pct"] || 0,
    risk_gov_pct: map["risk_gov_pct"] || 0,
    pilot_readiness_pct: map["pilot_readiness_pct"] || 0,
  };

  const payloadCol = map["payload"] || map["Payload"] || map["PAYLOAD"] || 0;
  const maxCol = Math.max(
    colCompany,
    colTs||0,
    payloadCol||0,
    ...Object.values(idxCols).filter(n=>n)
  );

  const values = sh.getRange(2, 1, lastRow-1, maxCol).getValues();
  const items = [];

  for(let i=0;i<values.length;i++){
    const row = i+2;
    const rowVals = values[i];
    if(String(rowVals[colCompany-1]||"").trim() !== company) continue;

    const it = {
      row,
      company,
      timestamp: colTs ? rowVals[colTs-1] : "",
      workplaces: idxCols.workplaces ? rowVals[idxCols.workplaces-1] : "",
      servers: idxCols.servers ? rowVals[idxCols.servers-1] : "",
      psi: idxCols.psi ? num_(rowVals[idxCols.psi-1]) : null,
      tech: idxCols.tech ? num_(rowVals[idxCols.tech-1]) : null,
      proc: idxCols.proc ? num_(rowVals[idxCols.proc-1]) : null,
      risk: idxCols.risk ? num_(rowVals[idxCols.risk-1]) : null,
      prob: idxCols.prob ? num_(rowVals[idxCols.prob-1]) : null,
      ready: idxCols.ready ? num_(rowVals[idxCols.ready-1]) : null,
      coiTotal: idxCols.coiTotal ? num_(rowVals[idxCols.coiTotal-1]) : null,
      recover: idxCols.recover ? num_(rowVals[idxCols.recover-1]) : null,
      bus: idxCols.bus ? num_(rowVals[idxCols.bus-1]) : null,
      obsol: idxCols.obsol ? num_(rowVals[idxCols.obsol-1]) : null,
      ai: idxCols.ai ? num_(rowVals[idxCols.ai-1]) : null,
      pain: idxCols.pain ? num_(rowVals[idxCols.pain-1]) : null,
      feas: idxCols.feas ? num_(rowVals[idxCols.feas-1]) : null,
      risk_grey_pct: idxCols.risk_grey_pct ? num_(rowVals[idxCols.risk_grey_pct-1]) : null,
      risk_lic_pct: idxCols.risk_lic_pct ? num_(rowVals[idxCols.risk_lic_pct-1]) : null,
      risk_ops_pct: idxCols.risk_ops_pct ? num_(rowVals[idxCols.risk_ops_pct-1]) : null,
      risk_gov_pct: idxCols.risk_gov_pct ? num_(rowVals[idxCols.risk_gov_pct-1]) : null,
      pilot_readiness_pct: idxCols.pilot_readiness_pct ? num_(rowVals[idxCols.pilot_readiness_pct-1]) : null,
    };

    if(payloadCol){
      const raw = rowVals[payloadCol-1];
      if(raw && typeof raw === "string"){
        const s = String(raw).trim();
        if(s && (s[0] === "{" || s[0] === "[")){
          try{
            const parsed = JSON.parse(s);
            const obj = (parsed && typeof parsed === "object" && !Array.isArray(parsed)) ? parsed : null;
            if(obj){
              if(it.psi===null && obj.psi2Score!==undefined) it.psi = num_(obj.psi2Score);
              if(it.tech===null && obj.techIndex!==undefined) it.tech = num_(obj.techIndex);
              if(it.proc===null && obj.procIndex!==undefined) it.proc = num_(obj.procIndex);
              if(it.risk===null && obj.riskIndex!==undefined) it.risk = num_(obj.riskIndex);
              if(it.prob===null && obj.probIndex!==undefined) it.prob = num_(obj.probIndex);
              if(it.ready===null && obj.readyIndex!==undefined) it.ready = num_(obj.readyIndex);
              if(it.coiTotal===null && obj.coiTotal!==undefined) it.coiTotal = num_(obj.coiTotal);
              if(it.recover===null && obj.recoverIndex!==undefined) it.recover = num_(obj.recoverIndex);
              if(it.bus===null && obj.busIndex!==undefined) it.bus = num_(obj.busIndex);
              if(it.obsol===null && obj.obsolIndex!==undefined) it.obsol = num_(obj.obsolIndex);
              if(it.ai===null && obj.aiIndex!==undefined) it.ai = num_(obj.aiIndex);
              if(it.pain===null && obj.painIndex!==undefined) it.pain = num_(obj.painIndex);
              if(it.feas===null && obj.feasScore!==undefined) it.feas = num_(obj.feasScore);
            }
          }catch(_e){}
        }
      }
    }

    items.push(it);
  }

  items.sort((a,b)=>{
    const ta = a.timestamp ? new Date(a.timestamp).getTime() : NaN;
    const tb = b.timestamp ? new Date(b.timestamp).getTime() : NaN;
    if(!isNaN(ta) && !isNaN(tb)) return ta - tb;
    return (a.row||0) - (b.row||0);
  });

  const sliced = items.length > limit ? items.slice(items.length-limit) : items;
  return {ok:true, company, items:sliced, total:items.length, version:CFG.VERSION};
}

function output_(obj, callback){
  if(callback){
    const js = callback + "(" + JSON.stringify(obj) + ");";
    return ContentService.createTextOutput(js).setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

/* ---------- Storage helpers ---------- */

function getSpreadsheet_(){
  if(!CFG.SPREADSHEET_ID) throw new Error("CFG.SPREADSHEET_ID is not set");
  return SpreadsheetApp.openById(CFG.SPREADSHEET_ID);
}

function getDataSheet_(){
  const ss = getSpreadsheet_();
  const sheets = ss.getSheets();
  for(const sh of sheets){
    if(sh.getSheetId && sh.getSheetId() === Number(CFG.SHEET_GID)) return sh;
  }
  return ss.getActiveSheet() || ss.getSheets()[0];
}

function getHeaderMap_(sheet){
  // We rely on the header row as the schema. If your sheet was created with a shorter header row,
  // indices/finance blocks won’t be saved. So we auto-restore the full schema from MASTER_HEADERS.
  const masterLen = (typeof MASTER_HEADERS !== "undefined" && MASTER_HEADERS && MASTER_HEADERS.length) ? MASTER_HEADERS.length : 0;

  // Read current header row as far as possible (use max columns to avoid getLastColumn truncation)
  const maxCols = Math.max(sheet.getMaxColumns(), masterLen || 0);
  if(maxCols < 1) return {headers:[], map:{}};

  let headers = sheet.getRange(1,1,1,maxCols).getValues()[0].map(h=>String(h||"").trim());

  // Trim trailing empties
  while(headers.length && !headers[headers.length-1]) headers.pop();

  // If header row is empty OR shorter than master OR misses any master columns → restore master schema
  const needsRestore = (headers.length === 0) || (masterLen && headers.length < masterLen) || (masterLen && MASTER_HEADERS.some((h,i)=>String(headers[i]||"").trim() !== String(h||"").trim()));
  if(needsRestore && masterLen){
    // Ensure sheet has enough columns
    const curMax = sheet.getMaxColumns();
    if(curMax < masterLen){
      sheet.insertColumnsAfter(curMax, masterLen - curMax);
    }
    sheet.getRange(1,1,1,masterLen).setValues([MASTER_HEADERS]);
    headers = MASTER_HEADERS.slice();
  }

  const map = {};
  headers.forEach((h,i)=>{ if(h) map[h]=i+1; });
  return {headers, map};
}

function normalizeKey_(s){
  return String(s||"")
    .toLowerCase()
    .replace(/[\s_\-\.]+/g, "")
    .replace(/[\(\)\[\]\{\}:;,\/\\'"`~!@#$%^&*+=?|<>]/g, "")
    .trim();
}

function buildNormPayloadMap_(payload){
  const out = {};
  for(const k in payload){
    if(!payload.hasOwnProperty(k)) continue;
    out[normalizeKey_(k)] = payload[k];
  }
  return out;
}

function getValueForHeader_(header, payload, payloadLower, payloadNorm){
  if(!header) return "";
  const h = String(header);
  const lk = h.toLowerCase();
  const nk = normalizeKey_(h);

  if(payload.hasOwnProperty(h)) return payload[h];
  if(payloadLower.hasOwnProperty(lk)) return payloadLower[lk];
  if(payloadNorm.hasOwnProperty(nk)) return payloadNorm[nk];

  const aliases = HEADER_ALIASES_[lk] || HEADER_ALIASES_[nk] || [];
  for(const a of aliases){
    const an = normalizeKey_(a);
    if(payload.hasOwnProperty(a)) return payload[a];
    if(payloadLower.hasOwnProperty(a.toLowerCase())) return payloadLower[a.toLowerCase()];
    if(payloadNorm.hasOwnProperty(an)) return payloadNorm[an];
  }

  return "";
}

const HEADER_ALIASES_ = {
  "sales manager": ["sales_manager","salesmanager","fio_sales_manager","фио sales manager","менеджер по продажам","sales"],
  "salesmanager": ["sales_manager","sales manager","фио sales manager","менеджер по продажам","sales"],
  "pre-sale менеджер": ["presale_manager","presalemanager","pre_sale_manager","пресейл","presale"],
  "presale manager": ["presale_manager","presalemanager","pre_sale_manager","пресейл","presale"],
  "presalemanager": ["presale_manager","presale manager","pre-sale менеджер","пресейл","presale"],
  "источники данных": ["sources","data_sources","datasources","источник данных","source"],
  "sources": ["источники данных","data_sources","datasources","источник данных","source"],
  "timestamp": ["Timestamp","Время","Дата","created_at","createdAt"],
};

function findTimestampCol_(headers, map){
  return map["timestamp"] || map["Timestamp"] || map["Время"] || map["Дата"] || map["created_at"] || map["createdAt"] || 0;
}

/* ---------- Actions ---------- */

function handleSearch_(p){
  const q = String(p.q||"").trim().toLowerCase();
  const limit = Math.max(1, Math.min(50, Number(p.limit||10)));
  const sh = getDataSheet_();
  const {headers, map} = getHeaderMap_(sh);
  const colCompany = map["company"] || map["Компания"] || 1;
  const colTs = findTimestampCol_(headers, map);
  const lastRow = sh.getLastRow();
  const items = [];

  if(lastRow >= 2){
    const nameVals = sh.getRange(2,colCompany,lastRow-1,1).getValues();
    const tsVals = colTs ? sh.getRange(2,colTs,lastRow-1,1).getValues() : null;
    const seen = {};

    for(let i=nameVals.length-1;i>=0;i--){
      const name = String(nameVals[i][0]||"").trim();
      if(!name) continue;
      const low = name.toLowerCase();
      if(q && low.indexOf(q) < 0) continue;
      if(seen[low]) continue;
      seen[low] = true;
      const row = i+2;
      const timestamp = tsVals ? String(tsVals[i][0]||"") : "";
      items.push({company:name, row, timestamp});
      if(items.length >= limit) break;
    }
  }

  return {ok:true, items, version:CFG.VERSION};
}

function handleGet_(p){
  const sh = getDataSheet_();
  const {headers, map} = getHeaderMap_(sh);
  const lastRow = sh.getLastRow();
  if(lastRow < 2) return {ok:false, error:"No data", version:CFG.VERSION};

  const rowParam = p.row ? Number(p.row) : 0;
  const company = String(p.company||"").trim();

  let foundRow = -1;
  const colCompany = map["company"] || map["Компания"] || 1;

  if(rowParam && rowParam >= 2 && rowParam <= lastRow){
    foundRow = rowParam;
  } else {
    if(!company) return {ok:false, error:"company or row is required", version:CFG.VERSION};
    const names = sh.getRange(2,colCompany,lastRow-1,1).getValues();
    for(let i=names.length-1;i>=0;i--){
      if(String(names[i][0]||"").trim() === company){ foundRow = i+2; break; }
    }
    if(foundRow < 0) return {ok:false, error:"Not found", version:CFG.VERSION};
  }

  const rowVals = sh.getRange(foundRow,1,1,headers.length).getValues()[0];
  const item = {};
  for(let i=0;i<headers.length;i++){
    const h = headers[i];
    if(!h) continue;
    item[h] = rowVals[i];
    item[String(h).toLowerCase()] = rowVals[i];
  }

  const payloadCol = map["payload"] || map["Payload"] || map["PAYLOAD"] || 0;
  if(payloadCol){
    const raw = rowVals[payloadCol-1];
    if(raw && typeof raw === "string"){
      const s = String(raw).trim();
      if(s && (s[0] === "{" || s[0] === "[")){
        try{
          const parsed = JSON.parse(s);
          item._payload_parsed = parsed;
          if(parsed && typeof parsed === "object" && !Array.isArray(parsed)){
            for(const k in parsed){
              if(!parsed.hasOwnProperty(k)) continue;
              if(item[k] === undefined || item[k] === null || item[k] === "") item[k] = parsed[k];
              const lk = String(k).toLowerCase();
              if(item[lk] === undefined || item[lk] === null || item[lk] === "") item[lk] = parsed[k];
            }
          }
        }catch(_e){}
      }
    }
  }

  item._row = foundRow;
  item._company = item.company || item["Компания"] || company || "";

  // Backward compatible: some pages expect `payload` key
  return {ok:true, payload:item, item, version:CFG.VERSION};
}

function handleSave_(e){
  let payloadStr = "";
  if(e && e.parameter && e.parameter.payload) payloadStr = String(e.parameter.payload);
  if(!payloadStr && e && e.postData && e.postData.contents){
    const c = String(e.postData.contents||"");
    const m = /payload=([^&]+)/.exec(c);
    if(m) payloadStr = decodeURIComponent(m[1]);
    else payloadStr = c;
  }
  payloadStr = String(payloadStr||"").trim();
  if(!payloadStr) return {ok:false, error:"payload is empty", version:CFG.VERSION};

  let payload;
  try{ payload = JSON.parse(payloadStr); }
  catch(_e){ return {ok:false, error:"payload is not valid JSON", version:CFG.VERSION}; }

  const company = String(payload.company || payload.Компания || "").trim();
  if(!company) return {ok:false, error:"payload.company is required", version:CFG.VERSION};

  const sh = getDataSheet_();
  const {headers, map} = getHeaderMap_(sh);
  if(!headers.length) return {ok:false, error:"Sheet header row is empty", version:CFG.VERSION};

  const colCompany = map["company"] || map["Компания"] || 1;

  const payloadLower = {};
  for(const k in payload){ if(payload.hasOwnProperty(k)) payloadLower[String(k).toLowerCase()] = payload[k]; }
  const payloadNorm = buildNormPayloadMap_(payload);

  const lastRow = sh.getLastRow();
  let latestRow = -1;
  if(lastRow >= 2){
    const names = sh.getRange(2,colCompany,lastRow-1,1).getValues();
    for(let i=names.length-1;i>=0;i--){
      if(String(names[i][0]||"").trim() === company){ latestRow = i+2; break; }
    }
  }

  const upsert = !!payload._upsert;
  const targetRow = (upsert && latestRow > 0) ? latestRow : (lastRow + 1);

  const row = new Array(headers.length).fill("");

  const nowIso = new Date().toISOString();
  const tsCol = findTimestampCol_(headers, map);
  if(tsCol){
    const idx = tsCol-1;
    const current = getValueForHeader_(headers[idx], payload, payloadLower, payloadNorm);
    if(!current) row[idx] = nowIso;
  }

  for(let i=0;i<headers.length;i++){
    const h = headers[i];
    if(!h) continue;
    if(i === (colCompany-1)) { row[i] = company; continue; }

    let v = getValueForHeader_(h, payload, payloadLower, payloadNorm);
    if(v && (typeof v === "object")){
      try{ v = JSON.stringify(v); }catch(_e){}
    }
    if(typeof v === "boolean") v = v ? "Да" : "Нет";
    row[i] = (v === undefined || v === null) ? "" : v;
  }

  sh.getRange(targetRow,1,1,headers.length).setValues([row]);
  return {ok:true, saved:true, row:targetRow, company, upsert, version:CFG.VERSION};
}

function handleStats_(p){
  const sh = getDataSheet_();
  const {headers} = getHeaderMap_(sh);
  const lastRow = sh.getLastRow();
  const rows = [];
  if(lastRow >= 2 && headers.length){
    const data = sh.getRange(2,1,lastRow-1,headers.length).getValues();
    for(const r of data){
      const obj = {};
      for(let i=0;i<headers.length;i++){
        const h = headers[i];
        if(!h) continue;
        obj[h] = r[i];
        obj[String(h).toLowerCase()] = r[i];
      }
      obj.company = obj.company || obj["Компания"] || obj["company"];
      obj.timestamp = obj.timestamp || obj["timestamp"] || obj["Timestamp"] || obj["Время"] || obj["Дата"];
      obj.psi = valFirst_(obj, ["psi","psi2Score","psi2score","psi_total","psitotal","psiTotal","psi2Score"]);
      obj.ready = valFirst_(obj, ["ready","readyIndex","readyindex"]);
      obj.probability = valFirst_(obj, ["probability","probIndex","probindex"]);
      obj.risk = valFirst_(obj, ["risk","riskIndex","riskindex"]);
      obj.synthetic_count = valFirst_(obj, ["synthetic_count","syntheticCount","syntheticcount"]);
      obj.total_fields = valFirst_(obj, ["total_fields","totalFields","totalfields"]);
      rows.push(obj);
    }
  }
  const companiesSet = {};
  for(const r of rows){
    const c = String(r.company||"").trim();
    if(c) companiesSet[c]=1;
  }
  const summary = {runs: rows.length, companies: Object.keys(companiesSet).length};
  const latest_records = rows.slice(-50).reverse();
  const series = buildSeries_(rows);
  return {ok:true, version:CFG.VERSION, summary, runs:summary.runs, companies:summary.companies, rows, latest_records, series};
}

function valFirst_(obj, keys){
  for(const k of keys){
    if(obj[k] !== undefined && obj[k] !== null && obj[k] !== "") return obj[k];
  }
  return "";
}

function buildSeries_(rows){
  const buckets = new Map();
  for(const r of rows){
    const ts = r.timestamp ? new Date(r.timestamp) : null;
    const key = ts && !isNaN(ts.getTime())
      ? Utilities.formatDate(ts, Session.getScriptTimeZone(), "yyyy-MM-dd")
      : "unknown";
    if(!buckets.has(key)) buckets.set(key, {date:key, runs:0, psi:0, ready:0, prob:0, risk:0});
    const b = buckets.get(key);
    b.runs += 1;
    b.psi += Number(r.psi||0) || 0;
    b.ready += Number(r.ready||0) || 0;
    b.prob += Number(r.probability||0) || 0;
    b.risk += Number(r.risk||0) || 0;
  }
  const out = [];
  for(const [k,v] of buckets.entries()){
    out.push({
      date: v.date,
      runs: v.runs,
      psi_avg: v.runs ? (v.psi / v.runs) : 0,
      ready_avg: v.runs ? (v.ready / v.runs) : 0,
      probability_avg: v.runs ? (v.prob / v.runs) : 0,
      risk_avg: v.runs ? (v.risk / v.runs) : 0
    });
  }
  out.sort((a,b)=> String(a.date).localeCompare(String(b.date)));
  return out;
}
